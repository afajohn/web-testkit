 Quick Reference for locating Selectors:
 To find...,Use this Selector Format:
 - An ID — inspect(document.querySelector('#elementID'))
 - A Class — inspect(document.querySelector('.className'))
 - A specific Attribute — "inspect(document.querySelector('[data-testid=""value""]'))"

 ##To highlight them selector/element:
 (function() {
    const el = document.querySelector('.d-md-flex > button:nth-child(1)');
    if (el) {
        el.style.outline = '5px solid red';
        el.style.outlineOffset = '2px';
        el.scrollIntoView({ behavior: 'smooth', block: 'center' });
        console.log("Found it! The element is now highlighted in red.");
        inspect(el);
    } else {
        console.error("Could not find the element with that selector.");
    }
})();

(function() {
    // 1. PUT THE TEXT YOU ARE LOOKING FOR HERE
    const searchFor = "Group Singles"; 

    // 2. Find any link that contains that text (ignoring case/spaces)
    const target = Array.from(document.querySelectorAll('a')).find(el => 
        el.textContent.toLowerCase().includes(searchFor.toLowerCase().trim())
    );

    if (target) {
        // Highlighting for your visual confirmation
        target.style.setProperty('outline', '8px solid red', 'important');
        target.style.setProperty('background-color', 'yellow', 'important');
        target.scrollIntoView({ behavior: 'smooth', block: 'center' });

        // DEBUG DATA FOR YOU (The "Answer" to give the devs)
        console.log("%c --- QA REPORT DATA ---", "color: red; font-weight: bold; font-size: 14px;");
        console.log("TEXT FOUND:", target.innerText.trim());
        console.log("BROKEN URL:", target.getAttribute('href'));
        console.log("LOCATION:", target.closest('div, section, table, nav').className || "No class on parent");
        
        // Jumps your Elements tab to the right spot
        inspect(target);
    } else {
        console.error(`COULD NOT FIND: "${searchFor}". Check the spelling or if it's inside an iframe.`);
    }
})();

#Frame target command:
(function() {
    // Look for common video containers or the iframe itself
    const videoTargets = [
        'iframe[src*="youtube"]',
        'iframe[src*="video"]',
        '.video-container',
        '#movie_player'
    ];
    
    let found = false;
    
    videoTargets.forEach(selector => {
        const el = document.querySelector(selector);
        if (el && !found) {
            el.style.border = '10px solid red';
            el.style.boxShadow = '0 0 50px red';
            el.scrollIntoView({ behavior: 'smooth', block: 'center' });
            console.log("Target found: " + selector);
            inspect(el);
            found = true;
        }
    });

    if (!found) {
        console.error("Still can't see it. Try scrolling the video into view manually, then run this again.");
    }
})();
